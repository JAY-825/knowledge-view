# git常用指令文档

本文整理 Git 在日常开发中的高频操作指令，覆盖账号配置与查看、SSH 密钥生成与查看、本地与远程分支操作，以及提交、合并、回滚、标签、别名等实用场景。所有命令在 Windows PowerShell 环境可直接使用（部分 ssh-agent 命令需要以管理员身份运行）。

# 账号配置与查看

  • 初始化与设置
# 设置全局用户名与邮箱（影响所有仓库）
git config --global user.name "你的名字"
git config --global user.email "you@example.com"

# 设置当前仓库的用户名与邮箱（仅影响当前仓库）
git config user.name "项目内名字"
git config user.email "project@example.com"

# 设置默认主分支名称
git config --global init.defaultBranch main

# Windows 推荐换行符策略（避免 CRLF/Unix 差异）
git config --global core.autocrlf true

# 配置拉取策略（按需选择）
# 仅快进
git config --global pull.ff only
# 或使用变基
git config --global pull.rebase true

# SSH 密钥生成与查看

  • 生成密钥
# 推荐使用 ed25519 算法（更快更安全）
ssh-keygen -t ed25519 -C "you@example.com"
# 交互式选择保存路径，通常为：C:\\Users\\<你的用户名>\\.ssh\\id_ed25519

# 备用：RSA 4096 位
ssh-keygen -t rsa -b 4096 -C "you@example.com"

  • 启动 ssh-agent 并添加密钥（Windows）
# 以管理员权限启用并启动服务
Get-Service ssh-agent | Set-Service -StartupType Automatic
Start-Service ssh-agent

# 添加私钥（PowerShell 路径写法）
ssh-add $env:USERPROFILE\.ssh\id_ed25519

# 查看已加载的密钥
essh-add -l
# 移除指定密钥
essh-add -d $env:USERPROFILE\.ssh\id_ed25519
# 移除全部密钥
essh-add -D

# 远程仓库与克隆

# 从远程克隆仓库
git clone git@github.com:user/repo.git

# 初始化本地仓库，并绑定远程
git init
git remote add origin git@github.com:user/repo.git

# 查看远程列表与地址
git remote -v

# 更新远程地址
git remote set-url origin git@github.com:user/new-repo.git

# 重命名/移除远程
git remote rename origin upstream
git remote remove upstream

# 分支操作（本地）

# 查看本地/远程/全部分支
git branch
git branch -r
git branch -a

# 创建并切换分支（推荐 switch）
git switch -c feature/login
# 旧用法（仍可用）
git checkout -b feature/login

# 切换分支
git switch main

# 重命名当前分支
git branch -m new-name
# 重命名其他分支
git branch -m old-name new-name

# 删除本地分支（未合并需 -D）
git branch -d feature/login
git branch -D feature/login

# 查看分支的上游跟踪状态
git branch -vv

# 分支操作（远程）

# 推送并建立上游跟踪
git push -u origin feature/login

# 拉取远程分支并建立本地跟踪
git fetch origin
git switch --track origin/feature/login
# 或：
git checkout -t origin/feature/login

# 删除远程分支
git push origin --delete feature/login
# 等价写法：
git push origin :feature/login

# 同步远程更改（含清理已删除分支）
git fetch -p
# 或显式清理
git remote prune origin

# 提交/查看/日志

# 查看状态
git status

# 选择性暂存（交互式）
git add -p
# 暂存全部改动
git add .

# 提交
git commit -m "feat: 登录流程"
# 修改上一条提交信息/内容（谨慎使用）
git commit --amend

# 差异对比
git diff            # 工作区与暂存区
git diff --staged   # 暂存区与最新提交

# 日志（常用美化别名见下文）
git log --oneline --graph --decorate --all

# 合并与变基

# 合并分支到当前分支
git merge feature/login
# 禁用快进（保留合并节点）
git merge --no-ff feature/login
# 压缩合并为一次提交
git merge --squash feature/login

# 变基到最新主分支（将当前分支提交“搬家”到 main 之上）
git fetch origin
git rebase origin/main
# 交互式变基（重排/压缩提交）
git rebase -i HEAD~5

# 冲突处理后继续
git add <files>
git merge --continue
# 或变基继续
git rebase --continue
# 中止操作
git merge --abort
git rebase --abort

# 暂存与回滚

# 储藏工作区改动（不含未跟踪文件）
git stash push -m "WIP: 登录表单"
# 包含未跟踪文件
git stash push -u -m "WIP: 含新文件"

# 查看/应用/弹出
git stash list
git stash apply stash@{0}
git stash pop stash@{0}
# 删除某条或全部
git stash drop stash@{0}
git stash clear

# 回滚提交（生成反向提交，安全）
git revert <commit>

# 重置到某提交（谨慎，可能重写历史）
# 保留改动到暂存区（软重置）
git reset --soft <commit>
# 保留改动到工作区（默认混合）
git reset --mixed <commit>
# 丢弃改动（硬重置，危险）
git reset --hard <commit>

# 恢复文件到指定版本（新命令）
# 丢弃工作区改动
git restore <path>
# 将文件从暂存区移除
git restore --staged <path>

# 清理未跟踪文件/目录（危险）
git clean -fd

# 标签（版本标记）

# 轻量标签
git tag v1.0.0
# 附注标签（包含作者/时间/说明）
git tag -a v1.0.0 -m "release 1.0.0"

# 推送单个或全部标签
git push origin v1.0.0
git push origin --tags

# 删除标签（本地/远程）
git tag -d v1.0.0
git push origin --delete tag v1.0.0

# 常用别名（提高效率）

# 基础别名
git config --global alias.co checkout
git config --global alias.br branch
git config --global alias.st status
git config --global alias.cm "commit -m"

# 精简图形化日志（推荐）
git config --global alias.lg "log --graph --pretty=format:'%C(auto)%h%C(reset) - %s %C(dim)%an, %ar%C(reset) %C(auto)%d' --abbrev-commit --date=relative"

# 常见场景速查

# 首次推送并建立上游
git push -u origin main

# 将本地分支关联到远程已有分支
git fetch origin
git branch --set-upstream-to=origin/feature/login feature/login
# 或：
git switch --track origin/feature/login

# 更新并同步远程清理
git fetch -p

# 获取最新主分支并快速同步到当前分支
git fetch origin && git merge --ff-only origin/main
# 或保持线性历史
git fetch origin && git rebase origin/main
