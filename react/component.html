# 组件（React）

组件是 React 应用的核心构建块。React 应用由一个个独立、可复用的组件组成，每个组件负责页面中的一部分 UI。

React 组件分为两种：函数组件（推荐）和 类组件。

# 一、函数组件

函数组件是定义组件的最简单方式，就是一个返回 JSX 的 JavaScript 函数。

// 定义函数组件
function Welcome() {
  return <h1>Hello, World!</h1>;
}

// 箭头函数写法
const Welcome = () => {
  return <h1>Hello, World!</h1>;
};

// 使用组件
function App() {
  return (
    <div>
      <Welcome />
      <Welcome />
      <Welcome />
    </div>
  );
}

# 函数组件 vs 类组件对比

特性 函数组件 类组件
写法 函数，返回 JSX 继承 React.Component，实现 render()
状态管理 useState Hook this.state + this.setState()
生命周期 useEffect Hook 生命周期方法（componentDidMount 等）

# 五、组件生命周期

# 类组件生命周期

import React from 'react';

class Timer extends React.Component {
  constructor(props) {
    super(props);
    this.state = { seconds: 0 };
  }

  // 组件挂载后
  componentDidMount() {
    this.interval = setInterval(() => {
      this.setState((prev) => ({ seconds: prev.seconds + 1 }));
    }, 1000);
  }

  // 组件更新后
  componentDidUpdate(prevProps, prevState) {
    console.log(`秒数更新：${prevState.seconds} → ${this.state.seconds}`);
  }

  // 组件卸载前
  componentWillUnmount() {
    clearInterval(this.interval);
  }

  render() {
    return <div>已运行：{this.state.seconds} 秒</div>;
  }
}

# 函数组件模拟生命周期（useEffect）

import { useState, useEffect } from 'react';

function Timer() {
  const [seconds, setSeconds] = useState(0);

  useEffect(() => {
    // 相当于 componentDidMount
    const interval = setInterval(() => {
      setSeconds((s) => s + 1);
    }, 1000);

    // 相当于 componentWillUnmount
    return () => {
      clearInterval(interval);
    };
  }, []); // 空依赖数组 = 只在挂载和卸载时执行

  // 相当于 componentDidUpdate
  useEffect(() => {
    console.log(`秒数变了：${seconds}`);
  }, [seconds]); // 依赖 seconds，seconds 变化时执行

  return <div>已运行：{seconds} 秒</div>;
}

# 八、列表渲染

function TodoList() {
  const [todos, setTodos] = useState([
    { id: 1, text: '学习 React', done: true },
    { id: 2, text: '写项目', done: false },
    { id: 3, text: '部署上线', done: false },
  ]);

  const toggleTodo = (id) => {
    setTodos((prev) =>
      prev.map((todo) =>
        todo.id === id ? { ...todo, done: !todo.done } : todo
      )
    );
  };

  const deleteTodo = (id) => {
    setTodos((prev) => prev.filter((todo) => todo.id !== id));
  };

  return (
    <ul>
      {todos.map((todo) => (
        <li key={todo.id}>
          <span
            style={{
              textDecoration: todo.done ? 'line-through' : 'none',
              cursor: 'pointer',
            }}
            onClick={() => toggleTodo(todo.id)}
          >
            {todo.text}
          </span>
          <button onClick={() => deleteTodo(todo.id)}>删除</button>
        </li>
      ))}
    </ul>
  );
}

# 4. Context 跨层级通信

import { createContext, useContext, useState } from 'react';

// 创建 Context
const ThemeContext = createContext('light');

// 提供者组件
function App() {
  const [theme, setTheme] = useState('light');

  return (
    <ThemeContext.Provider value={theme}>
      <div>
        <button onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}>
          切换主题
        </button>
        <Toolbar />
      </div>
    </ThemeContext.Provider>
  );
}

// 中间组件
function Toolbar() {
  return <ThemedButton />;
}

// 消费者组件
function ThemedButton() {
  const theme = useContext(ThemeContext);

  const style = {
    backgroundColor: theme === 'dark' ? '#333' : '#fff',
    color: theme === 'dark' ? '#fff' : '#333',
    padding: '8px 16px',
  };

  return <button style={style}>当前主题：{theme}</button>;
}

import { useState } from 'react';

function ControlledForm() {
  const [formData, setFormData] = useState({
    username: '',
    password: '',
    remember: false,
    city: 'beijing',
  });

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;

    e.preventDefault();

      <input
        name="username"
        value={formData.username}
        onChange={handleChange}
        placeholder="用户名"
      />
      <input
        name="password"
        type="password"
        value={formData.password}
        onChange={handleChange}
        placeholder="密码"
      />
      <label>
        <input
          name="remember"
          type="checkbox"
          checked={formData.remember}
          onChange={handleChange}
        />
        记住我
      </label>
      <select name="city" value={formData.city} onChange={handleChange}>
        <option value="beijing">北京</option>
        <option value="shanghai">上海</option>
        <option value="shenzhen">深圳</option>
      </select>
      <button type="submit">提交</button>
    </form>
  );
}

# 十一、高阶组件（HOC）

高阶组件是一个函数，接收一个组件作为参数，返回一个新的增强组件。

import { useState } from 'react';

// 高阶组件：添加 loading 功能
function withLoading(WrappedComponent) {
  return function WithLoadingComponent({ isLoading, ...props }) {
    if (isLoading) {
      return <div className="loading">加载中...</div>;
    }
    return <WrappedComponent {...props} />;
  };
}

// 普通组件
function UserList({ users }) {
  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}

// 增强后的组件
const UserListWithLoading = withLoading(UserList);

// 使用
function App() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(false);

  return (
    <div>
      <UserListWithLoading isLoading={loading} users={users} />
      <button onClick={async () => {
        setLoading(true);
        // 模拟请求
        const res = await fetch('/api/users');
        const data = await res.json();
        setUsers(data);
        setLoading(false);
      }}>
        加载用户
      </button>
    </div>
  );
}

Render Props 是一种通过 props 传递渲染逻辑的组件复用模式。

import { useState, useEffect } from 'react';

// 提供鼠标位置的组件
function MouseTracker({ render }) {
  const [position, setPosition] = useState({ x: 0, y: 0 });

  useEffect(() => {
    const handleMouseMove = (e) => {
      setPosition({ x: e.clientX, y: e.clientY });
    };
    window.addEventListener('mousemove', handleMouseMove);
    return () => window.removeEventListener('mousemove', handleMouseMove);
  }, []);

  return render(position);
}

// 使用 Render Props
function App() {
  return (
    <div>
      <h1>鼠标位置</h1>
      <MouseTracker
        render={({ x, y }) => (
          <p>
            鼠标坐标：({x}, {y})
          </p>
        )}
      />

      {/* 不同渲染方式 */}
      <MouseTracker
        render={({ x, y }) => (
          <div
            style={{
              position: 'fixed',
              left: x + 10,
              top: y + 10,
              pointerEvents: 'none',
            }}
          >
            光标
          </div>
        )}
      />
    </div>
  );
}

# 十三、组合（Composition）vs 继承

React 推荐使用组合而非继承来实现组件复用。

// 通过组合构建复杂组件
function Dialog({ title, children, footer }) {
  return (
    <div className="dialog-overlay">
      <div className="dialog">
        <div className="dialog-header">
          <h2>{title}</h2>
        </div>
        <div className="dialog-body">{children}</div>
        {footer && <div className="dialog-footer">{footer}</div>}
      </div>
    </div>
  );
}

function App() {
  return (
    <Dialog
      title="确认删除"

        <div>
          <button>取消</button>
          <button className="primary">确认</button>
        </div>
      }
    >
      <p>确定要删除这条记录吗？此操作不可撤销。</p>
    </Dialog>
  );
}

# 十四、组件拆分原则

// 不推荐：一个组件做所有事情
function UserPage() {
  const [users, setUsers] = useState([]);
  const [search, setSearch] = useState('');
  const [sortBy, setSortBy] = useState('name');

  useEffect(() => {
    fetch('/api/users').then((res) => res.json()).then(setUsers);
  }, []);

  const filtered = users
    .filter((u) => u.name.includes(search))
    .sort((a, b) => (a[sortBy] > b[sortBy] ? 1 : -1));

  return (
    <div>
      <input value={search} onChange={(e) => setSearch(e.target.value)} />
      <select value={sortBy} onChange={(e) => setSortBy(e.target.value)}>
        <option value="name">按姓名</option>
        <option value="age">按年龄</option>
      </select>
      <ul>
        {filtered.map((u) => (
          <li key={u.id}>{u.name} - {u.age}岁</li>
        ))}
      </ul>
    </div>
  );
}

// 推荐：拆分为多个职责单一的组件
function UserPage() {
  const [users, setUsers] = useState([]);
  const [search, setSearch] = useState('');
  const [sortBy, setSortBy] = useState('name');

  useEffect(() => {
    fetch('/api/users').then((res) => res.json()).then(setUsers);
  }, []);

  const filtered = users
    .filter((u) => u.name.includes(search))
    .sort((a, b) => (a[sortBy] > b[sortBy] ? 1 : -1));

  return (
    <div>
      <SearchBar search={search} onSearchChange={setSearch} />
      <SortSelect sortBy={sortBy} onSortChange={setSortBy} />

}

function SearchBar({ search, onSearchChange }) {
  return (
    <input
      value={search}
      onChange={(e) => onSearchChange(e.target.value)}

  );
}

function SortSelect({ sortBy, onSortChange }) {
  return (
    <select value={sortBy} onChange={(e) => onSortChange(e.target.value)}>
      <option value="name">按姓名</option>
      <option value="age">按年龄</option>
    </select>
  );
}

function UserList({ users }) {
  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name} - {user.age}岁</li>
      ))}
    </ul>
  );
}

# 总结

  • 优先使用函数组件 + Hooks，而非类组件
  • Props 用于父组件向子组件传递数据，子组件只读不可修改
  • State 用于管理组件内部可变数据，通过 useState 管理
  • 组件通信：父传子用 Props，子传父用回调，兄弟组件用状态提升，跨层级用 Context
  • 表单优先使用受控组件，数据由 React state 控制
  • 组件复用优先使用组合模式，避免继承
  • 保持组件职责单一，及时拆分复杂组件
  • 高阶组件和 Render Props 是组件逻辑复用的经典模式
