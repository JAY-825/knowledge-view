# Generator 与生成器函数

Generator（生成器）是 JavaScript 中一种“可暂停/可恢复执行”的函数机制。生成器函数（function*）执行后不会立刻跑完，而是返回一个生成器对象（generator object）。你可以通过 next() 一步步推动它执行，并在执行过程中用 yield 产出值，实现惰性计算、按需迭代、状态机、协程等能力。

# 一、核心概念

# 1. 生成器函数（generator function）

使用 function* 声明：

function* gen() {
  yield 1;
  yield 2;
  return 3;
}

调用 gen() 返回生成器对象：

const it = gen();

# 2. 生成器对象（generator object）

生成器对象同时满足两个协议：

  1. 迭代器协议：有 next() 方法，返回 { value, done }
  2. 可迭代协议：有 [Symbol.iterator]()，通常返回它自己

因此它可以直接用于 for...of、展开运算符、Array.from 等消费场景。

const it = gen();
console.log(typeof it.next); // function
console.log(it[Symbol.iterator]() === it); // true

# 二、语法与执行模型

# 1. yield：暂停并产出

  • yield 表达式 会把值产出给调用方，并暂停函数执行
  • 下一次调用 next() 会从暂停点继续执行
function* gen() {
  console.log('start');
  yield 'A';
  console.log('after A');
  yield 'B';
  console.log('after B');
}

const it = gen();
it.next(); // start -> { value:'A', done:false }
it.next(); // after A -> { value:'B', done:false }
it.next(); // after B -> { value:undefined, done:true }

# 2. next(value)：把值“送回”生成器

next(x) 的参数会作为“上一条 yield 表达式的结果”返回到生成器内部。

function* gen() {
  const name = yield '请输入姓名';
  const age = yield `你好 ${name}，请输入年龄`;
  return { name, age };
}

const it = gen();
console.log(it.next()); // { value:'请输入姓名', done:false }
console.log(it.next('Alice')); // { value:'你好 Alice，请输入年龄', done:false }
console.log(it.next(18)); // { value:{ name:'Alice', age:18 }, done:true }

# 3. return(value)：提前结束

it.return(x) 会让生成器立刻结束，并返回 { value: x, done: true }。

function* gen() {
  yield 1;
  yield 2;
  yield 3;
}

const it = gen();
console.log(it.next()); // 1
console.log(it.return('stop')); // { value:'stop', done:true }
console.log(it.next()); // { value:undefined, done:true }

# 4. throw(error)：向生成器内部抛异常

it.throw(err) 会在生成器当前暂停的 yield 位置抛出异常，生成器内部可以用 try/catch 捕获。

function* gen() {
  try {
    yield 1;
    yield 2;
  } catch (e) {
    yield `caught: ${e.message}`;
  }
  return 'done';
}

const it = gen();
console.log(it.next()); // 1
console.log(it.throw(new Error('boom'))); // caught: boom
console.log(it.next()); // done

# 5. yield*：委托给另一个可迭代对象

yield* iterable 会把控制权交给另一个可迭代对象，把它产出的值“透传”出来。

function* inner() {
  yield 'x';
  yield 'y';
  return 'innerReturn';
}

function* outer() {
  yield 'start';
  const r = yield* inner();
  yield `inner returned: ${r}`;
  yield 'end';
}

console.log(Array.from(outer()));
// ['start','x','y','inner returned: innerReturn','end']

# 三、Generator 相关内置对象与原型

JavaScript 没有一个标准的全局 Generator 构造函数供你 new Generator()。通常你会接触到：

  1. 生成器函数本身：function* () {}
  2. 生成器对象：调用生成器函数的返回值
  3. 生成器函数的构造器：Object.getPrototypeOf(function*(){}).constructor（通常表现为 GeneratorFunction）
function* g() {}
const it = g();

console.log(it.constructor.name); // 通常是 GeneratorFunction 或空字符串（环境差异）
console.log(typeof it.next); // function
console.log(typeof it.throw); // function
console.log(typeof it.return); // function

# 四、与迭代协议结合使用

# 1. for...of、展开、Array.from

function* gen() {
  yield 1;
  yield 2;
  yield 3;
}

for (const x of gen()) {
  console.log(x);
}

console.log([...gen()]); // [1,2,3]
console.log(Array.from(gen())); // [1,2,3]

# 五、常用模式与案例

# 1. 惰性序列：斐波那契（可控数量）

function* fibonacci(limit) {
  let a = 0;
  let b = 1;
  for (let i = 0; i < limit; i++) {
    yield b;
    [a, b] = [b, a + b];
  }
}

console.log([...fibonacci(10)]);

# 2. 惰性处理管道：filter/map（按需计算）

function* filter(iterable, predicate) {
  for (const x of iterable) {
    if (predicate(x)) yield x;
  }
}

function* map(iterable, mapper) {
  for (const x of iterable) {
    yield mapper(x);
  }
}

function* range(start, end) {
  for (let i = start; i <= end; i++) yield i;
}

const result = map(
  filter(range(1, 100), x => x % 2 === 0),
  x => x * x
);

console.log([...result].slice(0, 10));

# 4. 遍历树：深度优先（DFS）

function* dfs(node) {
  yield node.value;
  for (const child of node.children ?? []) {
    yield* dfs(child);
  }
}

const tree = {
  value: 'A',
  children: [
    { value: 'B', children: [{ value: 'D', children: [] }] },
    { value: 'C', children: [{ value: 'E', children: [] }, { value: 'F', children: [] }] }
  ]
};

console.log([...dfs(tree)]);

# 5. 简易协程：用 Generator 驱动 Promise（学习用）

这个模式用于理解“把异步步骤写成同步流程”。现代项目一般直接使用 async/await，但 Generator 的思路在一些库的设计中仍有价值。

function run(genFn) {
  const it = genFn();

  function step(nextFn, arg) {
    let r;
    try {
      r = nextFn.call(it, arg);
    } catch (e) {
      return Promise.reject(e);
    }

    if (r.done) return Promise.resolve(r.value);

    return Promise.resolve(r.value).then(
      v => step(it.next, v),
      e => step(it.throw, e)
    );
  }

  return step(it.next);
}

function wait(ms, value) {
  return new Promise(resolve => setTimeout(() => resolve(value), ms));
}

run(function* () {
  const a = yield wait(100, 1);
  const b = yield wait(100, 2);
  return a + b;
}).then(console.log);

# 六、与 async/await、Async Generator 的关系

# 1. Generator vs async function

  • Generator：手动驱动（next/throw/return），可暂停与恢复，适合表达“可迭代/可逐步推进”的流程
  • async function：返回 Promise，由运行时自动驱动 await，适合表达“异步串行流程”

# 2. Async Generator（异步生成器）

异步生成器使用 async function*，配合 for await...of 消费，常用于流式/分页数据。

async function* stream() {
  yield await Promise.resolve(1);
  yield await Promise.resolve(2);
}

(async () => {
  for await (const x of stream()) {
    console.log(x);
  }
})();

# 七、最佳实践与注意事项

  1. 生成器适合“按需产出”的场景：大数据遍历、分页、懒加载处理、复杂状态机
  2. 不要在无限生成器上盲目展开：[...infinite()] 会卡死或撑爆内存
  3. 明确结束条件：无限序列需要在消费端 break 或者在生成器内部实现上限
  4. 使用 yield* 简化组合：把复杂逻辑拆成多个小生成器复用
  5. for...of 不会拿到 return 值：如果你需要 return 值，改用手动 next() 驱动或在 yield 中产出
