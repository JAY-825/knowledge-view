# JavaScript 常用设计模式详解

# 一、单例模式 (Singleton Pattern)

# 用处

确保一个类/对象只有一个实例，并提供全局访问点。

# 使用场景

  1. 全局状态管理（如 Redux Store）
  2. 数据库连接池
  3. 日志记录器
  4. 配置管理器
  5. 缓存系统

# 代码示例

// ES6 实现：通过静态属性缓存实例
class Logger {
  constructor() {
    if (Logger.instance) return Logger.instance;
    this.logs = [];
    Logger.instance = this;
  }

  log(message) {
    const timestamp = new Date().toISOString();
    this.logs.push({ message, timestamp });
    console.log(`[${timestamp}] ${message}`);
  }

  getLogs() {
    return this.logs;
  }
}

// 使用
const logger1 = new Logger();
const logger2 = new Logger();
console.log(logger1 === logger2); // true

logger1.log('First log');
logger2.log('Second log');
console.log(logger1.getLogs().length); // 2

// 闭包实现（更贴近“模块单例”）
const ConfigManager = (function() {
  let instance;
  const config = {};
  
  function init() {
    return {
      set(key, value) {
        config[key] = value;
      },
      get(key) {
        return config[key];
      },
      getAll() {
        return { ...config };
      }
    };
  }
  
  return {
    getInstance() {
      if (!instance) {
        instance = init();
      }
      return instance;
    }
  };
})();

// 使用
const config1 = ConfigManager.getInstance();
const config2 = ConfigManager.getInstance();
config1.set('apiUrl', 'https://api.example.com');
console.log(config2.get('apiUrl')); // https://api.example.com

# 二、工厂模式 (Factory Pattern)

# 用处

创建对象而不暴露创建逻辑，通过一个公共接口来创建不同类型的对象。

# 使用场景

  1. UI组件创建（按钮、输入框等）
  2. 支付方式选择（支付宝、微信、银行卡）
  3. 通知类型（邮件、短信、推送）
  4. 文件解析器（JSON、XML、CSV）

# 代码示例

// 简单工厂模式
class Button {
  render() {
    console.log('Rendering button');
  }
}

class Input {
  render() {
    console.log('Rendering input');
  }
}

class UIFactory {
  static createElement(type) {
    switch(type) {
      case 'button':
        return new Button();
      case 'input':
        return new Input();
      default:
        throw new Error(`Unknown element type: ${type}`);
    }
  }
}

// 使用
const button = UIFactory.createElement('button');
button.render(); // Rendering button

// 工厂方法模式
class PaymentMethod {
  pay(amount) {
    throw new Error('Method must be implemented');
  }
}

class Alipay extends PaymentMethod {
  pay(amount) {
    console.log(`Paying ${amount} via Alipay`);
  }
}

class WechatPay extends PaymentMethod {
  pay(amount) {
    console.log(`Paying ${amount} via Wechat Pay`);
  }
}

class PaymentFactory {
  createPayment(type) {
    throw new Error('Factory method must be implemented');
  }
}

class AlipayFactory extends PaymentFactory {
  createPayment() {
    return new Alipay();
  }
}

class WechatPayFactory extends PaymentFactory {
  createPayment() {
    return new WechatPay();
  }
}

// 使用
const alipayFactory = new AlipayFactory();
const payment = alipayFactory.createPayment();
payment.pay(100); // Paying 100 via Alipay

# 三、观察者模式 (Observer Pattern)

# 用处

定义对象间的一对多依赖关系，当一个对象状态改变时，所有依赖它的对象都会收到通知。

# 使用场景

  1. 事件处理系统
  2. 数据绑定（如 Vue/React 响应式系统）
  3. 订阅式状态更新
  4. 实时聊天应用

# 代码示例

// 观察者模式实现
class Subject {
  constructor() {
    this.observers = [];
    this.state = null;
  }

  subscribe(observer) {
    this.observers.push(observer);
  }

  unsubscribe(observer) {
    this.observers = this.observers.filter(obs => obs !== observer);
  }

  notify() {
    this.observers.forEach(observer => observer.update(this.state));
  }

  setState(state) {
    this.state = state;
    this.notify();
  }
}

class Observer {
  constructor(name) {
    this.name = name;
  }

  update(state) {
    console.log(`${this.name} received update: ${state}`);
  }
}

// 使用
const weatherStation = new Subject();

const phoneDisplay = new Observer('Phone Display');
const tvDisplay = new Observer('TV Display');

weatherStation.subscribe(phoneDisplay);
weatherStation.subscribe(tvDisplay);

weatherStation.setState('Sunny');
// Phone Display received update: Sunny
// TV Display received update: Sunny

// 发布-订阅模式（通过事件中心解耦发布者与订阅者）
class EventBus {
  constructor() {
    this.events = {};
  }

  on(event, callback) {
    if (!this.events[event]) {
      this.events[event] = [];
    }
    this.events[event].push(callback);
  }

  off(event, callback) {
    if (!this.events[event]) return;
    this.events[event] = this.events[event].filter(cb => cb !== callback);
  }

  emit(event, data) {
    if (!this.events[event]) return;
    this.events[event].forEach(callback => callback(data));
  }
}

// 使用
const bus = new EventBus();

// 用户注册模块
bus.on('user:registered', (userData) => {
  console.log(`Sending welcome email to ${userData.email}`);
});

// 通知模块
bus.on('user:registered', (userData) => {
  console.log(`Creating notification for admin about new user ${userData.name}`);
});

// 触发事件
bus.emit('user:registered', {
  name: 'John Doe',
  email: 'john@example.com'
});

# 四、策略模式 (Strategy Pattern)

# 用处

定义一系列算法，封装每个算法，并使它们可以互相替换。

# 使用场景

  1. 表单验证规则
  2. 排序算法选择
  3. 支付方式计算
  4. 折扣策略

# 代码示例

// 策略模式实现
class DiscountStrategy {
  calculate(price) {
    throw new Error('Method must be implemented');
  }
}

class NoDiscount extends DiscountStrategy {
  calculate(price) {
    return price;
  }
}

class PercentageDiscount extends DiscountStrategy {
  constructor(percentage) {
    super();
    this.percentage = percentage;
  }

  calculate(price) {
    return price * (1 - this.percentage / 100);
  }
}

class FixedDiscount extends DiscountStrategy {
  constructor(amount) {
    super();
    this.amount = amount;
  }

  calculate(price) {
    return Math.max(0, price - this.amount);
  }
}

class ShoppingCart {
  constructor(discountStrategy = new NoDiscount()) {
    this.items = [];
    this.discountStrategy = discountStrategy;
  }

  addItem(item, price) {
    this.items.push({ item, price });
  }

  setDiscountStrategy(strategy) {
    this.discountStrategy = strategy;
  }

  getTotal() {
    const subtotal = this.items.reduce((sum, item) => sum + item.price, 0);
    return this.discountStrategy.calculate(subtotal);
  }
}

// 使用
const cart = new ShoppingCart();
cart.addItem('Book', 50);
cart.addItem('Shirt', 30);

console.log('Original total:', cart.getTotal()); // 80

// 应用10%折扣
cart.setDiscountStrategy(new PercentageDiscount(10));
console.log('With 10% discount:', cart.getTotal()); // 72

// 应用固定折扣
cart.setDiscountStrategy(new FixedDiscount(20));
console.log('With $20 discount:', cart.getTotal()); // 60

// 表单验证策略（支持规则参数）
const validationStrategies = {
  required: (value) => value ? null : 'This field is required',
  email: (value) => /\S+@\S+\.\S+/.test(value) ? null : 'Invalid email',
  minLength: (value, length) =>
    value.length >= length ? null : `Must be at least ${length} characters`,
  maxLength: (value, length) =>
    value.length <= length ? null : `Must be at most ${length} characters`
};

class Validator {
  constructor(strategies) {
    this.strategies = strategies;
  }

  validate(value, rules) {
    for (const rule of rules) {
      const { name, args = [] } = rule;
      const error = this.strategies[name]?.(value, ...args);
      if (error) return error;
    }
    return null;
  }
}

// 使用
const validator = new Validator(validationStrategies);
validator.validate('abc', [
  { name: 'required' },
  { name: 'minLength', args: [5] }
]);

# 五、装饰器模式 (Decorator Pattern)

# 用处

动态地为对象添加新功能，而不改变其结构。

# 使用场景

  1. 添加日志功能
  2. 缓存装饰器
  3. 权限验证
  4. 性能监控

# 代码示例

// 装饰器模式实现
class Coffee {
  cost() {
    return 5;
  }

  description() {
    return 'Simple coffee';
  }
}

// 装饰器基类
class CoffeeDecorator {
  constructor(coffee) {
    this.coffee = coffee;
  }

  cost() {
    return this.coffee.cost();
  }

  description() {
    return this.coffee.description();
  }
}

// 具体装饰器
class MilkDecorator extends CoffeeDecorator {
  cost() {
    return this.coffee.cost() + 2;
  }

  description() {
    return `${this.coffee.description()}, with milk`;
  }
}

class SugarDecorator extends CoffeeDecorator {
  cost() {
    return this.coffee.cost() + 1;
  }

  description() {
    return `${this.coffee.description()}, with sugar`;
  }
}

class WhippedCreamDecorator extends CoffeeDecorator {
  cost() {
    return this.coffee.cost() + 3;
  }

  description() {
    return `${this.coffee.description()}, with whipped cream`;
  }
}

// 使用
let myCoffee = new Coffee();
console.log(`${myCoffee.description()} costs $${myCoffee.cost()}`);

myCoffee = new MilkDecorator(myCoffee);
console.log(`${myCoffee.description()} costs $${myCoffee.cost()}`);

myCoffee = new SugarDecorator(myCoffee);
console.log(`${myCoffee.description()} costs $${myCoffee.cost()}`);

myCoffee = new WhippedCreamDecorator(myCoffee);
console.log(`${myCoffee.description()} costs $${myCoffee.cost()}`);

装饰器语法示例为传统实现，需要 Babel/TypeScript 的 experimentalDecorators 配置，当前 TC39 新提案语法与签名不同，使用时以项目配置为准。

function logger(target, key, descriptor) {
  const originalMethod = descriptor.value;
  
  descriptor.value = function(...args) {
    console.log(`Calling ${key} with arguments:`, args);
    const result = originalMethod.apply(this, args);
    console.log(`Result:`, result);
    return result;
  };
  
  return descriptor;
}

function cache(target, key, descriptor) {
  const originalMethod = descriptor.value;
  const cacheMap = new Map();
  
  descriptor.value = function(...args) {
    const cacheKey = JSON.stringify(args);
    
    if (cacheMap.has(cacheKey)) {
      console.log('Returning cached result');
      return cacheMap.get(cacheKey);
    }
    
    const result = originalMethod.apply(this, args);
    cacheMap.set(cacheKey, result);
    return result;
  };
  
  return descriptor;
}

class Calculator {
  @logger
  @cache
  heavyComputation(n) {
    console.log('Performing heavy computation...');
    let result = 0;
    for (let i = 0; i < n * 1000000; i++) {
      result += Math.random();
    }
    return result;
  }
}

// 使用
const calc = new Calculator();
calc.heavyComputation(5); // 执行计算
calc.heavyComputation(5); // 返回缓存结果

# 六、代理模式 (Proxy Pattern)

# 用处

为其他对象提供一种代理以控制对这个对象的访问。

# 使用场景

  1. 图片懒加载
  2. 缓存代理
  3. 权限控制
  4. 数据验证

# 代码示例

// 虚拟代理 - 图片懒加载
class RealImage {
  constructor(filename) {
    this.filename = filename;
    this.loadFromDisk();
  }

  loadFromDisk() {
    console.log(`Loading ${this.filename} from disk`);
  }

  display() {
    console.log(`Displaying ${this.filename}`);
  }
}

class ProxyImage {
  constructor(filename) {
    this.filename = filename;
    this.realImage = null;
  }

  display() {
    if (!this.realImage) {
      this.realImage = new RealImage(this.filename);
    }
    this.realImage.display();
  }
}

// 使用
const image1 = new ProxyImage('photo1.jpg');
const image2 = new ProxyImage('photo2.jpg');

// 图片不会立即加载
image1.display(); // 第一次调用时加载并显示
image1.display(); // 第二次直接显示（已加载）

// 缓存代理
class ExpensiveApi {
  fetchData(key) {
    console.log(`Fetching expensive data for: ${key}`);
    // 模拟网络请求延迟
    return new Promise(resolve => {
      setTimeout(() => {
        resolve(`Data for ${key}`);
      }, 1000);
    });
  }
}

class CachedApiProxy {
  constructor(api) {
    this.api = api;
    this.cache = new Map();
  }

  async fetchData(key) {
    if (this.cache.has(key)) {
      console.log(`Returning cached data for: ${key}`);
      return this.cache.get(key);
    }
    
    const data = await this.api.fetchData(key);
    this.cache.set(key, data);
    return data;
  }
}

// 使用
const api = new ExpensiveApi();
const cachedApi = new CachedApiProxy(api);

// 第一次调用，从API获取
cachedApi.fetchData('user-123').then(console.log);

// 第二次调用相同的key，从缓存获取
setTimeout(() => {
  cachedApi.fetchData('user-123').then(console.log);
}, 1500);

// 保护代理 - 权限控制
class SensitiveData {
  getData() {
    return 'Sensitive information';
  }
}

class DataProxy {
  constructor(user) {
    this.user = user;
    this.realData = new SensitiveData();
  }

  getData() {
    if (this.user.role === 'admin') {
      return this.realData.getData();
    } else {
      throw new Error('Access denied');
    }
  }
}

// 使用
const admin = { role: 'admin' };
const user = { role: 'user' };

const adminProxy = new DataProxy(admin);
console.log(adminProxy.getData()); // Sensitive information

const userProxy = new DataProxy(user);
try {
  console.log(userProxy.getData()); // Error: Access denied
} catch (error) {
  console.log(error.message);
}

# 七、适配器模式 (Adapter Pattern)

# 用处

将一个类的接口转换成客户端期望的另一个接口。

# 使用场景

  1. 新旧API兼容
  2. 第三方库适配
  3. 不同数据格式转换
  4. 接口标准化

# 代码示例

// 旧系统接口
class OldPaymentSystem {
  makePayment(amountInDollars) {
    console.log(`Processing payment of $${amountInDollars}`);
    return true;
  }
}

// 新系统期望的接口
class NewPaymentService {
  processPayment(amount, currency) {
    console.log(`Processing ${currency} ${amount}`);
    return { success: true };
  }
}

// 适配器
class PaymentAdapter {
  constructor(oldSystem) {
    this.oldSystem = oldSystem;
  }

  processPayment(amount, currency) {
    // 转换逻辑：欧元转美元
    if (currency === 'EUR') {
      const amountInDollars = amount * 1.1; // 简单转换率
      const result = this.oldSystem.makePayment(amountInDollars);
      return { success: result };
    }
    throw new Error(`Currency ${currency} not supported by old system`);
  }
}

// 使用
const oldSystem = new OldPaymentSystem();
const adapter = new PaymentAdapter(oldSystem);

// 新代码可以使用适配器接口
const result = adapter.processPayment(100, 'EUR');
console.log(result); // { success: true }

// 数据格式适配器示例
class XMLData {
  constructor(data) {
    this.data = data;
  }

  getXML() {
    return `<data>${this.data}</data>`;
  }
}

class JSONData {
  constructor(data) {
    this.data = data;
  }

  getJSON() {
    return JSON.stringify({ data: this.data });
  }
}

// 适配器：XML 转 JSON
class XMLToJSONAdapter {
  constructor(xmlData) {
    this.xmlData = xmlData;
  }

  getJSON() {
    const xml = this.xmlData.getXML();
    // 简单的XML到JSON转换逻辑
    const match = xml.match(/<data>(.*?)<\/data>/);
    if (match) {
      return JSON.stringify({ data: match[1] });
    }
    return JSON.stringify({ data: null });
  }
}

// 使用
const xmlData = new XMLData('Hello World');
const adapter2 = new XMLToJSONAdapter(xmlData);
console.log(adapter2.getJSON()); // {"data":"Hello World"}

# 八、组合模式 (Composite Pattern)

# 用处

将对象组合成树形结构以表示"部分-整体"的层次结构。

# 使用场景

  1. 文件系统
  2. UI组件树
  3. 组织结构图
  4. 菜单系统

# 代码示例

// 组合模式实现
class FileSystemComponent {
  constructor(name) {
    this.name = name;
  }

  add(component) {
    throw new Error('This method must be overridden');
  }

  remove(component) {
    throw new Error('This method must be overridden');
  }

  getSize() {
    throw new Error('This method must be overridden');
  }

  display(indent = 0) {
    throw new Error('This method must be overridden');
  }
}

// 叶子节点：文件
class File extends FileSystemComponent {
  constructor(name, size) {
    super(name);
    this.size = size;
  }

  getSize() {
    return this.size;
  }

  display(indent = 0) {
    console.log(`${'  '.repeat(indent)}📄 ${this.name} (${this.size} bytes)`);
  }
}

// 复合节点：文件夹
class Folder extends FileSystemComponent {
  constructor(name) {
    super(name);
    this.children = [];
  }

  add(component) {
    this.children.push(component);
  }

  remove(component) {
    const index = this.children.indexOf(component);
    if (index > -1) {
      this.children.splice(index, 1);
    }
  }

  getSize() {
    return this.children.reduce((total, child) => total + child.getSize(), 0);
  }

  display(indent = 0) {
    console.log(`${'  '.repeat(indent)}📁 ${this.name}`);
    this.children.forEach(child => child.display(indent + 1));
  }
}

// UI组件树示例
class UIComponent {
  constructor(name) {
    this.name = name;
    this.children = [];
  }

  add(component) {
    this.children.push(component);
  }

  remove(component) {
    const index = this.children.indexOf(component);
    if (index > -1) {
      this.children.splice(index, 1);
    }
  }

  render() {
    throw new Error('render method must be implemented');
  }
}

class Button extends UIComponent {
  constructor(name) {
    super(name);
  }

  render() {
    return `<button>${this.name}</button>`;
  }
}

class Div extends UIComponent {
  constructor(name) {
    super(name);
  }

  render() {
    const childrenHTML = this.children
      .map(child => child.render())
      .join('');
    return `<div>${childrenHTML}</div>`;
  }
}

// 使用文件系统示例
const root = new Folder('Root');
const documents = new Folder('Documents');
const music = new Folder('Music');

documents.add(file1);
documents.add(file2);
music.add(file3);

root.add(documents);
root.add(music);

root.display();
console.log(`Total size: ${root.getSize()} bytes`);

const container = new Div('container');
const header = new Div('header');
const button1 = new Button('Click me');
const button2 = new Button('Submit');

header.add(button1);
container.add(header);

# 九、状态模式 (State Pattern)

# 用处

允许对象在内部状态改变时改变其行为。

# 使用场景

  1. 订单状态（待付款、已发货、已完成等）
  2. 游戏角色状态（正常、中毒、眩晕等）
  3. 工作流状态
  4. 交通灯状态

# 代码示例

// 状态模式实现
class OrderState {
  constructor(order) {
    this.order = order;
  }

  next() {
    throw new Error('Method must be implemented');
  }

  previous() {
    throw new Error('Method must be implemented');
  }

  cancel() {
    throw new Error('Method must be implemented');
  }
}

class PendingState extends OrderState {
  next() {
    console.log('Processing payment...');
    this.order.setState(new PaidState(this.order));
  }

  cancel() {
    console.log('Order cancelled');
    this.order.setState(new CancelledState(this.order));
  }
}

class PaidState extends OrderState {
  next() {
    console.log('Shipping order...');
    this.order.setState(new ShippedState(this.order));
  }

  previous() {
    console.log('Reverting payment...');
    this.order.setState(new PendingState(this.order));
  }
}

class ShippedState extends OrderState {
  next() {
    console.log('Order delivered');
    this.order.setState(new DeliveredState(this.order));
  }

  previous() {
    console.log('Returning to warehouse...');
    this.order.setState(new PaidState(this.order));
  }
}

class DeliveredState extends OrderState {
  previous() {
    console.log('Cannot go back from delivered state');
  }
}

class CancelledState extends OrderState {
  previous() {
    console.log('Cannot go back from cancelled state');
  }
}

class Order {
  constructor() {
    this.state = new PendingState(this);
  }

  setState(state) {
    this.state = state;
    console.log(`Order state changed to: ${state.constructor.name}`);
  }

  next() {
    this.state.next();
  }

  previous() {
    this.state.previous();
  }

  cancel() {
    this.state.cancel();
  }
}

// 使用
const order = new Order();

order.next(); // Processing payment... -> PaidState
order.next(); // Shipping order... -> ShippedState
order.next(); // Order delivered -> DeliveredState

class TrafficLight {

    this.states = {
      RED: new RedLight(this),
      YELLOW: new YellowLight(this),
      GREEN: new GreenLight(this)

  }

  change() {
    this.currentState.change();
  }

  setState(state) {
    this.currentState = state;
  }

  getState() {
    return this.currentState.constructor.name;
  }
}

class LightState {
  constructor(light) {
    this.light = light;
  }

  change() {
    throw new Error('Method must be implemented');
  }

  change() {
    console.log('Green light - GO');
    this.light.setState(this.light.states.YELLOW);
  }
}

const trafficLight = new TrafficLight();

for (let i = 0; i < 6; i++) {
  trafficLight.change();
}

# 十、迭代器模式 (Iterator Pattern)

# 用处

提供一种方法顺序访问一个聚合对象中的各个元素，而又不暴露其内部表示。

# 使用场景

  1. 集合遍历
  2. 树形结构遍历
  3. 数据库结果集遍历
  4. 文件系统遍历

# 代码示例

// 自定义迭代器实现
class Collection {
  constructor(items = []) {
    this.items = items;
  }

  add(item) {
    this.items.push(item);
  }

  [Symbol.iterator]() {
    let index = 0;
    const items = this.items;
    
    return {
      next() {
        if (index < items.length) {
          return { value: items[index++], done: false };
        } else {
          return { done: true };
        }
      },
      
      // 可选：提前结束迭代
      return() {
        console.log('Iteration stopped early');
        return { done: true };
      }
    };
  }
}

// 使用
const collection = new Collection([1, 2, 3, 4, 5]);

for (const item of collection) {
  if (item > 3) break; // 会触发return方法
  console.log(item);
}

// 树形结构迭代器
class TreeNode {
  constructor(value) {
    this.value = value;
    this.children = [];
  }

  addChild(child) {
    this.children.push(child);
  }

  // 深度优先遍历
  *depthFirst() {
    yield this.value;
    for (const child of this.children) {
      yield* child.depthFirst();
    }
  }

  // 广度优先遍历
  *breadthFirst() {
    const queue = [this];
    while (queue.length > 0) {
      const node = queue.shift();
      yield node.value;
      queue.push(...node.children);
    }
  }
}

// 使用
const root = new TreeNode('Root');
const child1 = new TreeNode('Child 1');
const child2 = new TreeNode('Child 2');
const grandchild1 = new TreeNode('Grandchild 1');
const grandchild2 = new TreeNode('Grandchild 2');

child1.addChild(grandchild1);
child2.addChild(grandchild2);
root.addChild(child1);
root.addChild(child2);

console.log('Depth-first traversal:');
for (const value of root.depthFirst()) {
  console.log(value);
}

console.log('\nBreadth-first traversal:');
for (const value of root.breadthFirst()) {
  console.log(value);
}

// 范围迭代器
function* range(start, end, step = 1) {
  for (let i = start; i <= end; i += step) {
    yield i;
  }
}

// 使用
console.log('\nRange iteration:');
for (const num of range(1, 10, 2)) {
  console.log(num); // 1, 3, 5, 7, 9
}

// 斐波那契序列迭代器（可限制数量）
function* fibonacci(limit) {
  let [prev, curr] = [0, 1];
  let count = 0;
  
  while (count < limit) {
    yield curr;
    [prev, curr] = [curr, prev + curr];
    count++;
  }
}

// 使用
console.log('\nFibonacci sequence:');
for (const num of fibonacci(10)) {
  console.log(num);
}

# 总结

设计模式 主要用途 典型场景
单例模式 确保只有一个实例 全局状态、配置管理
工厂模式 创建对象封装 UI组件、支付方式
观察者模式 一对多依赖通知 事件系统、数据绑定
策略模式 算法封装替换 表单验证、折扣计算
装饰器模式 动态添加功能 日志、缓存、权限
代理模式 控制对象访问 懒加载、缓存、权限
适配器模式 接口转换 API兼容、数据转换
组合模式 树形结构管理 文件系统、UI组件树
状态模式 状态行为管理 订单状态、游戏状态
迭代器模式 集合遍历 数据遍历、树遍历

这些设计模式在 JavaScript 开发中非常实用，掌握它们可以帮助你编写更可维护、可扩展和高效的代码。在实际项目中，经常会出现多种模式组合使用的情况，理解每种模式的核心思想比死记硬背实现方式更重要。
