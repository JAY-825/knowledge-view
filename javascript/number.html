# Number

Number 是 JavaScript 处理数值的内置对象，既是构造函数也是一组与数值相关的静态属性与方法的载体。底层采用 IEEE‑754 双精度浮点表示，这会带来精度与舍入的特性与限制。

# 构造函数

  • Number(value)：将传入值转换为数值并返回原始类型（primitive）
  • new Number(value)：创建一个 Number 包装对象（不推荐在业务中使用）
Number('42');        // 42
Number(true);        // 1
Number(null);        // 0
Number(undefined);   // NaN

const obj = new Number(42);
typeof obj;          // 'object'
obj.valueOf();       // 42（解包为原始数值）

示例：使用 EPSILON 比较近似相等

function almostEqual(a, b, eps = Number.EPSILON) {
  return Math.abs(a - b) < eps;
}
almostEqual(0.1 + 0.2, 0.3); // true（浮点误差容忍）

# 静态方法

  • Number.isFinite(value)：仅当 value 为数值且有限返回 true
Number.isFinite(1);     // true
Number.isFinite(Infinity); // false
Number.isFinite('1');   // false（不会先转换）

  • Number.isInteger(value)：判断是否为整数
Number.isInteger(5);                       // true
Number.isInteger(5.0);                     // true（能用整数编码表示的浮点数也为 true）
Number.isInteger(NaN);                     // false
Number.isInteger(Infinity);                // false
Number.isInteger('5');                     // false

// 精度影响示例（接近安全整数的值）
Number.isInteger(5.0000000000000001);      // true（与 5 使用同一编码表示）
Number.isInteger(4500000000000000.1);      // true
Number.isInteger(4500000000000000.5);      // false

# 代码示例（综合）

// 判断整数与安全整数
function isValidId(n) {
  return Number.isInteger(n) && Number.isSafeInteger(n) && n > 0;
}

// 浮点加法的近似比较
function addEqual(a, b, expected) {
  const sum = a + b;
  return Math.abs(sum - expected) < Number.EPSILON;
}

isValidId(100);                 // true
isValidId(9007199254740992);    // false
addEqual(0.1, 0.2, 0.3);        // true

// 解析与格式化
const price = Number.parseFloat('19.9元');
price.toFixed(2); // '19.90'
price.toLocaleString('zh-CN', { style: 'currency', currency: 'CNY' });
